\section*{One-shot SITL launch}
\begin{itemize}\footnotesize
  \item \texttt{./waf configure --board sitl --debug}
  \item \texttt{./waf plane}
  \item \texttt{Tools/autotest/sim\_vehicle.py -v ArduPlane -f plane --console --map}
  \item Add \texttt{--gdb} to debug under gdb (M9, L1).
  \item Add \texttt{--no-mavproxy} for headless gdb-attach (L1 path).
\end{itemize}

\section*{Plane flight modes used in the labs}
\begin{itemize}\footnotesize
  \item \flmode{TAKEOFF} --- automated take-off (L3, L4).
  \item \flmode{FBWA} --- stabilised manual; sticks $\rightarrow$ desired roll/pitch (L4 Phase A).
  \item \flmode{FBWB} --- adds altitude hold; sticks $\rightarrow$ climb/descent rate (L4 Phase B).
  \item \flmode{CRUISE} --- altitude + heading hold.
  \item \flmode{RTL} --- return to launch; loiters at \texttt{RTL\_ALTITUDE} (no auto-land without DO\_LAND\_START).
  \item \flmode{AUTO} --- mission execution.
\end{itemize}

\section*{Critical ArduPlane invariants}
\begin{itemize}\footnotesize
  \item \textbf{Plane fast-loop = 50 Hz} (Copter = 400 Hz). Override via \texttt{SCHED\_LOOP\_RATE} (range 50--400, reboot required).
  \item \textbf{Cooperative scheduler}: no pre-emption; a 50 ms task overrun is silent.
  \item \textbf{\texttt{*\_cd} suffix} = centi-degrees, signed \texttt{int32\_t}. Canonical integer angle unit in vehicle code.
  \item \textbf{Never renumber \texttt{AP\_GROUPINFO} indices.} The index is baked into stored configs.
  \item \textbf{Never run \texttt{waf} with \texttt{sudo}}.
  \item \textbf{\texttt{AP\_HAL::millis()} wraps every $\sim$49 days; \texttt{micros()} every $\sim$71 min.} Use the subtract-and-let-it-wrap idiom.
\end{itemize}

\section*{HAL access patterns (the four)}
\begin{itemize}\footnotesize
  \item \texttt{\#include <AP\_HAL/AP\_HAL.h>} --- umbrella include.
  \item \texttt{extern const AP\_HAL::HAL\& hal;} --- the singleton handle.
  \item \texttt{AP\_HAL::millis()} / \texttt{micros()} --- free time functions.
  \item \texttt{hal.scheduler->delay(...)} --- the only \texttt{Scheduler} call most algorithms need.
\end{itemize}

\section*{\texttt{errorScore} formula in one line}
\footnotesize
Score = \texttt{MAX} of:\\
\quad GPS: \texttt{0.5(velTR + posTR)} (always),\\
\quad Alt: \texttt{hgtTR} (always),\\
\quad Air: \texttt{0.3 tasTR} (gated $\geq 2$ sensors + \texttt{EKF\_AFFINITY\_ARSP}),\\
\quad Mag: \texttt{0.3(magTR.x+y+z)} (gated \texttt{EKF\_AFFINITY\_MAG}).\\
Score $> \sim 1$ $\Rightarrow$ lane unhealthy. Periodic 10 s debounce; \texttt{checkLaneSwitch} bypasses it.

\section*{TECS one-line summary}
\footnotesize
$STE = h + V^2/(2g)$, $SEB = h - V^2/(2g)$. Throttle vs $STE$; pitch vs blend. Blend weight = \texttt{TECS\_SPDWEIGHT} (default 1.0 = 50/50; 0 = altitude only; 2 = speed only).

\section*{L1 one-line summary}
\footnotesize
\texttt{\_L1\_dist = MAX($1/\pi \cdot$ damping $\cdot$ period $\cdot$ groundSpeed, dist\_min)}. Defaults: \texttt{NAVL1\_PERIOD=17 s}, \texttt{NAVL1\_DAMPING=0.75}.

\section*{The four extraction-seam patterns (M10)}
\begin{enumerate}\footnotesize
  \item Library + stub HAL --- vendor source, mock the HAL surface they actually call. (L1, TECS, controllers.)
  \item Library + replace \texttt{AP\_Param} --- vendor source, swap config system. (When your config is JSON-backed.)
  \item Math/algorithm only + reimplement wiring --- read the algorithm, rewrite in your idioms. (EKF lane switch.)
  \item Black-box ArduPilot via MAVLink/DDS --- run unmodified, talk over the wire.
\end{enumerate}

\section*{Lab pass criteria}
\begin{itemize}\footnotesize
  \item \textbf{L1}: gdb stops at \texttt{Plane::ahrs\_update}; \texttt{millis()}>0; \texttt{loop\_rate}=50.
  \item \textbf{L2}: \texttt{MY\_PARAM 42.0} persists across SITL restart.
  \item \textbf{L3}: \texttt{"EKF3 lane switch"} STATUSTEXT within 30 s of glitch; \param{EV} in dataflash.
  \item \textbf{L4}: MAVExplorer plots show roll-rate change with \texttt{TCONST}; alt-track change with \texttt{PTCH\_DAMP}. Force-disarm via \texttt{param2=21196}.
  \item \textbf{L5}: \texttt{ctest} prints \texttt{100\% tests passed}.
\end{itemize}

\section*{When something is wrong --- triage in order}
\begin{enumerate}\footnotesize
  \item Stale binary? \texttt{ls -lt build/sitl/bin/arduplane}; rebuild if old.
  \item Stale SITL process? \texttt{pgrep arduplane}; kill before relaunching.
  \item ptrace blocked? \texttt{cat /proc/sys/kernel/yama/ptrace\_scope}; set 0 if locked.
  \item Lane switch not appearing? \texttt{param show EK3\_IMU\_MASK}; must be 3.
  \item Apt cascade on Ubuntu pre-release? Stale mirror, almost never the script. Switch to \texttt{archive.ubuntu.com}.
\end{enumerate}
